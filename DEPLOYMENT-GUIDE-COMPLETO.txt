==========================================
GUÍA COMPLETA DE DESPLIEGUE - SISTEMA GESTIÓN COLEGIO
==========================================

FECHA: 08/09/2025
SISTEMA: Ubuntu 24.04 LTS en VirtualBox
APLICACIÓN: NestJS + TypeORM + MySQL + Docker

==========================================
1. CONFIGURACIÓN INICIAL DEL SERVIDOR
==========================================

# Actualizar sistema
sudo apt update && sudo apt upgrade -y

# Instalar dependencias necesarias
sudo apt install -y curl wget git docker.io docker-compose python3 python3-pip

# Agregar usuario al grupo docker
sudo usermod -aG docker $USER

# Reiniciar sesión para aplicar cambios
# (cerrar y abrir terminal)

# Verificar instalación
docker --version
docker-compose --version

==========================================
2. CONFIGURACIÓN DE RED VIRTUALBOX
==========================================

IMPORTANTE: Para acceso desde red local

1. Cerrar máquina virtual
2. En VirtualBox Manager:
   - Seleccionar VM
   - Settings → Network → Adapter 1
   - Attached to: "Bridged Adapter"
   - Name: Seleccionar adaptador de red (WiFi/Ethernet)
   - Promiscuous Mode: "Allow All"
3. Reiniciar VM

# Verificar IP de red local
hostname -I
# Debe mostrar IP tipo 192.168.1.x

==========================================
3. CONFIGURACIÓN DE FIREWALL
==========================================

# Habilitar firewall y permitir puerto 443
sudo ufw allow 443/tcp
sudo ufw enable

# Verificar estado
sudo ufw status

==========================================
4. CLONAR Y CONFIGURAR PROYECTO
==========================================

# Clonar repositorio
git clone [URL_DEL_REPOSITORIO] gestion-backend-ar
cd gestion-backend-ar

# Configurar archivo .env
nano .env

# Contenido del .env:
DB_HOST=mysql
DB_PORT=3306
DB_USERNAME=gestion_user
DB_PASSWORD=gestion2024
DB_NAME=gestion_academica_ar
HTTPS_ENABLED=true
NODE_ENV=production
JWT_SECRET=tu_jwt_secret_aqui
TELEGRAM_BOT_TOKEN=tu_telegram_token_aqui
TELEGRAM_CHAT_ID=tu_chat_id_aqui

==========================================
5. CONFIGURACIÓN DE SSL
==========================================

# Crear directorio SSL
mkdir -p ssl

# Generar certificado autofirmado
openssl req -x509 -newkey rsa:4096 -keyout ssl/server.key -out ssl/server.crt -days 365 -nodes -subj "/C=PE/ST=Lima/L=Lima/O=Colegio/CN=localhost"

# Configurar permisos correctos para SSL
chmod 600 ssl/server.key
chmod 644 ssl/server.crt

# IMPORTANTE: Cambiar propietario al usuario del contenedor Docker
# Verificar el usuario del contenedor después de iniciarlo
docker compose up -d
docker compose exec app id
# Usar el UID y GID del comando anterior (ejemplo: 1001:65533)
sudo chown -R 1001:65533 ssl/

# Verificar certificados y permisos
ls -la ssl/

==========================================
6. CONFIGURACIÓN DE DIRECTORIOS
==========================================

# Crear directorios necesarios
mkdir -p logs
mkdir -p public/profiles/usuarios

# Verificar estructura
ls -la

==========================================
7. CONFIGURACIÓN DE DOCKER
==========================================

# Verificar docker-compose.yml
cat docker-compose.yml

# Verificar Dockerfile
cat Dockerfile

# Construir y ejecutar contenedores
docker compose up -d --build

# Verificar estado de contenedores
docker compose ps

==========================================
8. EJECUTAR MIGRACIONES
==========================================

# Verificar que la base de datos esté lista
docker compose exec mysql mysql -u root -p -e "SHOW DATABASES;"

# Ejecutar migraciones
docker compose exec app npm run typeorm migration:run

# Verificar que las tablas se crearon
docker compose exec mysql mysql -u root -p -e "USE gestion_academica_ar; SHOW TABLES;"

==========================================
9. VERIFICACIÓN DEL DESPLIEGUE
==========================================

# Verificar logs de la aplicación
docker compose logs app --tail=20

# Verificar que el puerto esté escuchando
sudo ss -tlnp | grep :443

# Probar desde el servidor
curl -k https://localhost:443/health

# Verificar IP del servidor
hostname -I

==========================================
10. CONFIGURACIÓN PARA ACCESO EXTERNO
==========================================

# IP del servidor (ejemplo: 192.168.1.109)
SERVER_IP=$(hostname -I | awk '{print $1}')
echo "IP del servidor: $SERVER_IP"

# URLs para acceso externo:
echo "URLs disponibles:"
echo "Health: https://$SERVER_IP:443/health"
echo "Login: https://$SERVER_IP:443/auth/login"
echo "API Docs: https://$SERVER_IP:443/api"
echo "Alumnos: https://$SERVER_IP:443/alumnos"

==========================================
11. CONFIGURACIÓN EN POSTMAN/FRONTEND
==========================================

# Configuración SSL en Postman:
# Settings → General → Desactivar "SSL certificate verification"

# Endpoint de login:

==========================================
13. SOLUCIÓN DE PROBLEMAS COMUNES
==========================================

# Problema: "No migrations are pending"
# Solución: Verificar que data-source.ts tenga:
# migrations: ['dist/database/migrations/*.js']

# Problema: SSL_ERROR_SYSCALL
# Solución: Usar -k en curl o desactivar verificación SSL

# Problema: No se puede acceder desde red externa
# Solución: Verificar configuración de red VirtualBox (Bridged Adapter)

# Problema: Puerto 443 no accesible
# Solución: Verificar firewall (sudo ufw allow 443/tcp)

# Problema: Contenedores no inician
# Solución: Verificar logs (docker compose logs)

==========================================
14. BACKUP Y RESTAURACIÓN
==========================================

# Backup de base de datos
docker compose exec mysql mysqldump -u root -p gestion_academica_ar > backup_$(date +%Y%m%d_%H%M%S).sql

# Restaurar base de datos
docker compose exec -i mysql mysql -u root -p gestion_academica_ar < backup_file.sql

# Backup de archivos de aplicación
tar -czf app_backup_$(date +%Y%m%d_%H%M%S).tar.gz gestion-backend-ar/

==========================================
NOTAS IMPORTANTES
==========================================

1. La aplicación está configurada para escuchar en 0.0.0.0:443 en producción
2. CORS está habilitado para todos los orígenes
3. HTTPS está configurado con certificado autofirmado
4. La base de datos se crea automáticamente con las migraciones
5. Los logs se guardan en el directorio ./logs/
6. Los certificados SSL están en ./ssl/
7. La aplicación es accesible desde cualquier dispositivo en la red local
