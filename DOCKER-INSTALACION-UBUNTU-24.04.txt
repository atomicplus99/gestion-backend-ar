# ==========================================
# INSTALACIÓN DOCKER EN UBUNTU 24.04.03 LTS
# ==========================================
# GUÍA COMPLETA - SOLO LO QUE FUNCIONA
# ==========================================

## PASO 1: ACTUALIZAR SISTEMA
sudo apt update
sudo apt install ca-certificates curl

## PASO 2: AGREGAR CLAVE GPG OFICIAL DE DOCKER
sudo install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /tmp/docker.asc
sudo mv /tmp/docker.asc /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

## PASO 3: AGREGAR REPOSITORIO OFICIAL DE DOCKER
echo "deb [arch=amd64 signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/ubuntu noble stable" | sudo tee /etc/apt/sources.list.d/docker.list

## PASO 4: ACTUALIZAR E INSTALAR DOCKER
sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io docker-compose-plugin docker-buildx-plugin

## PASO 5: AGREGAR USUARIO AL GRUPO DOCKER
sudo usermod -aG docker $USER

## PASO 6: VERIFICAR INSTALACIÓN
# Cerrar sesión y volver a entrar
exit
# (volver a conectarse)

# Verificar Docker
docker --version

# Verificar Docker Compose (sin guión)
docker compose version

# Probar Docker
sudo docker run hello-world

# ==========================================
# CONFIGURACIÓN VIRTUALBOX GUEST ADDITIONS
# ==========================================
# (Para habilitar copiar y pegar)

## PASO 1: DESCARGAR GUEST ADDITIONS
wget https://download.virtualbox.org/virtualbox/7.0.12/VBoxGuestAdditions_7.0.12.iso

## PASO 2: MONTAR ISO
sudo mount -o loop VBoxGuestAdditions_7.0.12.iso /mnt/cdrom

## PASO 3: INSTALAR GUEST ADDITIONS
sudo /mnt/cdrom/VBoxLinuxAdditions.run

## PASO 4: REINICIAR
sudo reboot

## PASO 5: DESPUÉS DEL REINICIO
# En VirtualBox: Dispositivos → Portapapeles compartido → Bidireccional
# Ahora funciona: Ctrl+Shift+V para pegar

# ==========================================
# COMANDOS DOCKER ÚTILES
# ==========================================

# Ver estado de contenedores
docker ps

# Ver todos los contenedores (incluyendo detenidos)
docker ps -a

# Ver imágenes
docker images

# Ver logs de un contenedor
docker logs nombre_contenedor

# Detener contenedor
docker stop nombre_contenedor

# Eliminar contenedor
docker rm nombre_contenedor

# Eliminar imagen
docker rmi nombre_imagen

# ==========================================
# DOCKER COMPOSE COMANDOS
# ==========================================

# Ejecutar servicios
docker compose up -d

# Detener servicios
docker compose down

# Ver logs
docker compose logs -f

# Ver estado
docker compose ps

# Construir imágenes
docker compose build

# ==========================================
# NOTAS IMPORTANTES
# ==========================================

# 1. Ubuntu 24.04.03 LTS funciona perfectamente con Docker
# 2. Usar "docker compose" (sin guión) en lugar de "docker-compose"
# 3. Guest Additions es necesario para copiar y pegar en VirtualBox
# 4. El repositorio "noble" es correcto para Ubuntu 24.04
# 5. Siempre reiniciar después de instalar Guest Additions
# 6. Verificar que el usuario esté en el grupo docker

# ==========================================
# VERIFICACIÓN FINAL
# ==========================================

# Estos comandos deben funcionar sin errores:
docker --version
docker compose version
sudo docker run hello-world
docker ps
