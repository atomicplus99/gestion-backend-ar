# ENDPOINT JUSTIFICACIONES - GUÍA PARA FRONTEND

## ENDPOINT PRINCIPAL
GET http://localhost:3000/detalle-justificaciones

## PARÁMETROS DE QUERY (TODOS OPCIONALES)
- codigo_alumno: string (ej: "2544822025")
- estado: enum ["PENDIENTE", "APROBADA", "RECHAZADA", "EN_REVISION"]
- tipo_justificacion: enum ["MEDICA", "FAMILIAR", "ACADEMICA", "PERSONAL", "EMERGENCIA"]
- fecha_desde: string formato "YYYY-MM-DD" (ej: "2025-08-01")
- fecha_hasta: string formato "YYYY-MM-DD" (ej: "2025-08-31")
- pagina: number (comienza en 1, ej: 1)
- elementos_por_pagina: number (ej: 10)

## EJEMPLOS DE USO
1. Todas las justificaciones:
   GET http://localhost:3000/detalle-justificaciones

2. Filtrar por alumno:
   GET http://localhost:3000/detalle-justificaciones?codigo_alumno=2544822025

3. Filtrar por estado:
   GET http://localhost:3000/detalle-justificaciones?estado=PENDIENTE

4. Filtrar por tipo y estado:
   GET http://localhost:3000/detalle-justificaciones?tipo_justificacion=MEDICA&estado=APROBADA

5. Con paginación:
   GET http://localhost:3000/detalle-justificaciones?pagina=1&elementos_por_pagina=5

6. Filtro por rango de fechas:
   GET http://localhost:3000/detalle-justificaciones?fecha_desde=2025-08-01&fecha_hasta=2025-08-31

## DTOs PARA FRONTEND (TYPESCRIPT)

### 1. RESPUESTA PRINCIPAL
interface JustificacionesResponse {
  statusCode: number;
  message: string;
  data: JustificacionItem[];
  total: number;
  paginacion: PaginacionInfo;
}

### 2. ITEM DE JUSTIFICACIÓN
interface JustificacionItem {
  id_justificacion: string;
  tipo_justificacion: "MEDICA" | "FAMILIAR" | "ACADEMICA" | "PERSONAL" | "EMERGENCIA";
  motivo: string;
  estado: "PENDIENTE" | "APROBADA" | "RECHAZADA" | "EN_REVISION";
  fecha_solicitud: string; // ISO date string
  fechas_de_justificacion: string[]; // Array de fechas en formato "DD-MM-YYYY"
  documentos_adjuntos?: string[];
  fecha_respuesta?: string | null; // ISO date string
  observaciones_solicitante?: string | null;
  alumno_solicitante: AlumnoInfo;
  responsable: ResponsableInfo;
  asistencias_creadas?: number;
}

### 3. INFORMACIÓN DEL ALUMNO
interface AlumnoInfo {
  id_alumno: string;
  codigo: string;
  nombre: string;
  apellido: string;
  nivel: string;
  grado: string;
  seccion: string;
  turno: TurnoInfo;
}

### 4. INFORMACIÓN DEL TURNO
interface TurnoInfo {
  id_turno: string;
  nombre_turno: string;
  hora_inicio: string; // formato "HH:MM:SS"
  hora_fin: string; // formato "HH:MM:SS"
}

### 5. INFORMACIÓN DEL RESPONSABLE
interface ResponsableInfo {
  tipo: "auxiliar" | "administrador" | "director" | "desconocido";
  id: string;
  nombre: string;
  apellido: string;
  correo_electronico: string;
}

### 6. INFORMACIÓN DE PAGINACIÓN
interface PaginacionInfo {
  pagina_actual: number;
  total_paginas: number;
  elementos_por_pagina: number;
  total_elementos: number;
  tiene_siguiente: boolean;
  tiene_anterior: boolean;
}

## PARÁMETROS DE QUERY (INTERFACE)
interface JustificacionesQueryParams {
  codigo_alumno?: string;
  estado?: "PENDIENTE" | "APROBADA" | "RECHAZADA" | "EN_REVISION";
  tipo_justificacion?: "MEDICA" | "FAMILIAR" | "ACADEMICA" | "PERSONAL" | "EMERGENCIA";
  fecha_desde?: string; // formato "YYYY-MM-DD"
  fecha_hasta?: string; // formato "YYYY-MM-DD"
  pagina?: number;
  elementos_por_pagina?: number;
}

## CÓDIGOS DE ERROR
- 400: Error en parámetros de filtrado o paginación
- 500: Error interno del servidor

## NOTAS IMPORTANTES
1. Las fechas de justificación vienen en formato "DD-MM-YYYY"
2. Las fechas de solicitud y respuesta vienen en formato ISO (UTC)
3. Los documentos adjuntos son opcionales y pueden ser null
4. La paginación comienza en página 1, no en 0
5. Si no se especifica elementos_por_pagina, usa un valor por defecto
6. El campo "asistencias_creadas" indica cuántas asistencias se generaron a partir de esta justificación
7. El campo "responsable" ahora incluye el tipo de usuario (auxiliar, administrador, director)
8. El campo "responsable.tipo" indica quién registró la justificación
